from typing import Optional, Dict, Any



class Money:

    """Immutable Value Object representing monetary amount and currency."""

    __slots__ = ("_amount", "_currency")



    def __init__(self, amount: float, currency: str = "USD") -> None:

        if amount < 0:

            raise ValueError("Amount cannot be negative")

        self._amount, self._currency = round(amount, 2), currency.upper()



    @property

    def amount(self) -> float:

        return self._amount



    @property

    def currency(self) -> str:

        return self._currency



    def add(self, other: "Money") -> "Money":

        if self._currency != other._currency:

            raise ValueError(f"Currency mismatch: {self._currency} vs {other._currency}")

        return Money(self._amount + other._amount, self._currency)



    def __eq__(self, other: object) -> bool:

        if not isinstance(other, Money):

            return False

        return self._amount == other._amount and self._currency == other._currency



    def __hash__(self) -> int:

        return hash((self._amount, self._currency))



    def __repr__(self) -> str:

        return f"Money({self._amount} {self._currency})"





class BankAccount:

    """Domain Entity representing a financial account with identity semantics."""

    __slots__ = ("_accountId", "_ownerName", "_balance", "_isActive")



    def __init__(self, accountId: str, ownerName: str, initialDeposit: Money) -> None:

        self._accountId, self._ownerName = accountId, ownerName

        self._balance, self._isActive = initialDeposit, True



    @property

    def accountId(self) -> str:

        return self._accountId



    @property

    def balance(self) -> Money:

        return self._balance



    def deposit(self, money: Money) -> None:

        if not self._isActive:

            raise ValueError("Cannot deposit into inactive account")

        self._balance = self._balance.add(money)



    def withdraw(self, money: Money) -> None:

        if not self._isActive:

            raise ValueError("Cannot withdraw from inactive account")

        if self._balance.amount < money.amount:

            raise ValueError("Insufficient balance")

        self._balance = Money(self._balance.amount - money.amount, self._balance.currency)



    def __eq__(self, other: object) -> bool:

        # Entity equality is strictly based on identity (ID), not state

        if not isinstance(other, BankAccount):

            return False

        return self._accountId == other._accountId



    def __hash__(self) -> int:

        return hash(self._accountId)



    def __repr__(self) -> str:

        return f"BankAccount(id={self._accountId}, owner='{self._ownerName}', balance={self._balance})"

Entity Identity vs Value Object Semantics

Use when: modeling domain models where identity matters (Accounts, Users) vs value equality (Money, Coordinates).

Architecture & Design Trace

Entities compare strictly by unique ID via __eq__ and __hash__. Value Objects are immutable, use __slots__, and compare by all attributes.

[Value Object: Money]             [Entity: BankAccount]
    (10 USD) == (10 USD)               id='acc-1', balance=10
         -> TRUE (Same Value)          id='acc-1', balance=50
                                            -> TRUE (Same Account ID)
    hash((10, 'USD'))                  hash('acc-1')
# Value Objects compare by state; Entities compare by ID

m1, m2 = Money(100.0, "USD"), Money(100.0, "USD")

assert m1 == m2

assert hash(m1) == hash(m2)



accA = BankAccount("ACC-101", "Alice", m1)

accB = BankAccount("ACC-101", "Alice Updated", Money(500.0, "USD"))

# True because accountId matches!

assert accA == accB

Pure Abstract Base Classes (ABCs)

Use when: defining rigid contracts and interfaces in machine coding to enforce polymorphism at instantiation time.

Architecture & Design Trace

Subclassing abc.ABC with @abstractmethod ensures Python prevents direct instantiation and guarantees all abstract methods are implemented.

         ▲

   StripePaymentGateway

from abc import ABC, abstractmethod

from typing import Dict, Any



class IPaymentGateway(ABC):

    """Abstract Base Class defining strict contract for payment processors."""



    @abstractmethod

    def processPayment(self, amount: float, currency: str) -> str:

        """Process charge and return unique transaction ID."""

        pass



    @abstractmethod

    def refundPayment(self, transactionId: str) -> bool:

        """Refund a previous transaction."""

        pass



class StripePaymentGateway(IPaymentGateway):

    def __init__(self, apiKey: str) -> None:

        self.apiKey = apiKey



    def processPayment(self, amount: float, currency: str) -> str:

        # Concrete implementation

        return f"stripe_tx_{int(amount * 100)}"



    def refundPayment(self, transactionId: str) -> bool:

        return True

Multiple Inheritance & C3 Linearization (MRO)

Use when: combining mixins or multiple interfaces without hitting the diamond dependency or uncooperative super() trap.

Architecture & Design Trace

Python resolves inheritance hierarchies using C3 Linearization via ClassName.mro(). Using cooperative super().__init__(**kwargs) guarantees every base class constructor runs exactly once in deterministic sequence.

         object
          ▲    ▲
          │    │
BaseWorker     LoggingMixin
          ▲    ▲
          │    │
      AuditedWorker
MRO: AuditedWorker -> LoggingMixin -> BaseWorker -> object
class BaseWorker:

    def __init__(self, workerId: str, **kwargs) -> None:

        self.workerId = workerId

        super().__init__(**kwargs)



    def execute(self) -> None:

        print(f"BaseWorker({self.workerId}) executing")



class LoggingMixin:

    def __init__(self, logPrefix: str = "LOG", **kwargs) -> None:

        self.logPrefix = logPrefix

        super().__init__(**kwargs)



    def execute(self) -> None:

        print(f"[{self.logPrefix}] Starting execution...")

        super().execute()

        print(f"[{self.logPrefix}] Execution completed.")



class AuditedWorker(LoggingMixin, BaseWorker):

    """Multiple inheritance resolved via C3 Linearization."""

    def __init__(self, workerId: str, **kwargs) -> None:

        super().__init__(workerId=workerId, logPrefix="AUDIT", **kwargs)

Cooperative Mixin Classes

        UserProfile

import json

from typing import Dict, Any



class JsonSerializableMixin:

    """Cooperative Mixin adding zero-cost JSON serialization to domain models."""

    def toJson(self) -> str:

        return json.dumps(self.toDict(), indent=2)



    def toDict(self) -> Dict[str, Any]:

        result = {}

        for attr in getattr(self, "__slots__", self.__dict__.keys()):

            val = getattr(self, attr)

            # Strip leading underscore for clean API export

            cleanKey = attr[1:] if attr.startswith("_") else attr

            result[cleanKey] = val.toDict() if hasattr(val, "toDict") else val

        return result



class UserProfile(JsonSerializableMixin):

    __slots__ = ("_userId", "_email", "_role")



    def __init__(self, userId: str, email: str, role: str) -> None:

        self._userId, self._email, self._role = userId, email, role



user = UserProfile("u-123", "alice@example.com", "ENGINEER")

print(user.toJson())

from typing import Dict, List, Protocol, Optional

from dataclasses import dataclass

@dataclass(frozen=True, slots=True)

class OrderItem:

    itemId: str

    price: float

    quantity: int



@dataclass

class Order:

    orderId: str

    userId: str

    items: List[OrderItem]

    totalAmount: float = 0.0



# 2. INTERFACE / CONTRACT (ISP & DIP)

class IOrderRepository(Protocol):

    def save(self, order: Order) -> None: ...

    def findById(self, orderId: str) -> Optional[Order]: ...



class IPaymentProcessor(Protocol):

    def charge(self, userId: str, amount: float) -> str: ...



# 3. OPEN/CLOSED DISCOUNT STRATEGY (OCP)

class IDiscountStrategy(ABC):

    @abstractmethod

    def calculateDiscount(self, order: Order) -> float: ...



class PercentageDiscount(IDiscountStrategy):

    def __init__(self, percent: float) -> None:

        self.percent = percent



    def calculateDiscount(self, order: Order) -> float:

        rawTotal = sum(item.price * item.quantity for item in order.items)

        return rawTotal * (self.percent / 100.0)



# 4. SERVICE ORCHESTRATOR (DIP & SRP)

class OrderFulfillmentService:

    """Orchestrates checkout strictly depending on abstractions (DIP)."""

    def __init__(

        self,

        orderRepo: IOrderRepository,

        paymentProcessor: IPaymentProcessor,

        discountStrategy: IDiscountStrategy

    ) -> None:

        self.orderRepo = orderRepo

        self.paymentProcessor = paymentProcessor

        self.discountStrategy = discountStrategy



    def processOrder(self, order: Order) -> str:

        subtotal = sum(i.price * i.quantity for i in order.items)

        discount = self.discountStrategy.calculateDiscount(order)

        order.totalAmount = max(0.0, subtotal - discount)



        txId = self.paymentProcessor.charge(order.userId, order.totalAmount)

        self.orderRepo.save(order)

        return txId

Open/Closed Principle (OCP)

Use when: adding new business rules, discount strategies, or payment providers without editing existing core classes.

Architecture & Design Trace

Use the Strategy Pattern combined with a class registry instead of chained if/elif statements that mutate every sprint.

                                               ├── PercentageDiscount
                                               └── BogoDiscount

from abc import ABC, abstractmethod

from typing import Dict, Type



class IPromotionStrategy(ABC):

    @abstractmethod

    def applyDiscount(self, amount: float) -> float: ...



class FlatDiscount(IPromotionStrategy):

    def __init__(self, flatDeduction: float) -> None:

        self.flatDeduction = flatDeduction

    def applyDiscount(self, amount: float) -> float:

        return max(0.0, amount - self.flatDeduction)



class BogoDiscount(IPromotionStrategy):

    def applyDiscount(self, amount: float) -> float:

        return amount * 0.5  # Buy one get one free



class PromotionRegistry:

    """OCP-Compliant Strategy Registry: Add new strategies without modifying caller!"""

    _strategies: Dict[str, Type[IPromotionStrategy]] = {}



    @classmethod

    def register(cls, promoCode: str, strategyCls: Type[IPromotionStrategy]) -> None:

        cls._strategies[promoCode.upper()] = strategyCls



    @classmethod

    def get(cls, promoCode: str) -> Optional[Type[IPromotionStrategy]]:

        return cls._strategies.get(promoCode.upper())



# Register dynamically

PromotionRegistry.register("FLAT20", FlatDiscount)

PromotionRegistry.register("BOGO", BogoDiscount)

Harness standard method annotations (@property, @classmethod, @dataclass) and build custom thread-safe decorators.

import functools

import threading

import time

from typing import Callable, Any, TypeVar



R = TypeVar("R")



def synchronized(method: Callable[..., R]) -> Callable[..., R]:

    """Decorator ensuring method executes with instance's reentrant lock."""

    @functools.wraps(method)

    def wrapper(self, *args: Any, **kwargs: Any) -> R:

        lock = getattr(self, "_lock", None)

        if lock is None:

            # Lazy initialize RLock if not defined

            self._lock = threading.RLock()

            lock = self._lock

        with lock:

            return method(self, *args, **kwargs)

    return wrapper



def retry(maxAttempts: int = 3, delaySeconds: float = 0.1) -> Callable:

    """Decorator that retries an operation on transient failure."""

    def decorator(func: Callable[..., R]) -> Callable[..., R]:

        @functools.wraps(func)

        def wrapper(*args: Any, **kwargs: Any) -> R:

            attempts, lastErr = 0, None

            while attempts < maxAttempts:

                try:

                    return func(*args, **kwargs)

                except Exception as err:

                    attempts += 1

                    lastErr = err

                    time.sleep(delaySeconds * (2 ** (attempts - 1)))

            raise RuntimeError(f"Failed after {maxAttempts} attempts") from lastErr

        return wrapper

    return decorator



class ThreadSafeMetrics:

    def __init__(self) -> None:

        self._lock = threading.RLock()

        self._counters = {}



    @synchronized

    def increment(self, metricName: str, delta: int = 1) -> int:

        self._counters[metricName] = self._counters.get(metricName, 0) + delta

        return self._counters[metricName]



    @synchronized

    def get(self, metricName: str) -> int:

        return self._counters.get(metricName, 0)

class BankCard:

    def __init__(self, cardNumber: str, cvv: str) -> None:

        self.cardNumber = cardNumber  # Invokes property setter for validation

        self.cvv = cvv



    @property

    def cardNumber(self) -> str:

        """Masked representation of sensitive card number."""

        return f"****-****-****-{self._cardNumber[-4:]}"



    @cardNumber.setter

    def cardNumber(self, value: str) -> None:

        sanitized = value.replace("-", "").strip()

        if len(sanitized) != 16 or not sanitized.isdigit():

            raise ValueError("Card number must be exactly 16 digits")

        self._cardNumber = sanitized



    @property

    def cvv(self) -> str:

        return self._cvv



    @cvv.setter

    def cvv(self, value: str) -> None:

        if len(value) != 3 or not value.isdigit():

            raise ValueError("CVV must be 3 digits")

        self._cvv = value

@classmethod receives cls and returns cls(...), allowing subclasses to inherit factories cleanly. @staticmethod has no state access.

         ▼
[Class Method receives cls]
   ├── Parses ISO strings

from datetime import datetime

from typing import Dict, Any



class TimeInterval:

    def __init__(self, startEpoch: int, endEpoch: int) -> None:

        if startEpoch >= endEpoch:

            raise ValueError("Start epoch must precede end epoch")

        self.startEpoch, self.endEpoch = startEpoch, endEpoch



    @classmethod

    def fromIsoStrings(cls, startIso: str, endIso: str) -> "TimeInterval":

        """Alternative Constructor: Parses ISO 8601 timestamps."""

        fmt = "%Y-%m-%dT%H:%M:%SZ"

        start = int(datetime.strptime(startIso, fmt).timestamp())

        end = int(datetime.strptime(endIso, fmt).timestamp())

        return cls(start, end)

import threading



class BankLedger:

    def __init__(self) -> None:

        # Use RLock instead of Lock when methods invoke other locked methods

        self._lock = threading.RLock()

        self._balances = {}



    def getBalance(self, accountId: str) -> float:

        with self._lock:

            return self._balances.get(accountId, 0.0)



    def deposit(self, accountId: str, amount: float) -> None:

        with self._lock:

            # Reentrancy: getBalance() acquires the same lock without deadlocking!

            current = self.getBalance(accountId)

            self._balances[accountId] = current + amount



    def transfer(self, fromAcc: str, toAcc: str, amount: float) -> bool:

        with self._lock:

            if self.getBalance(fromAcc) < amount:

                return False

            self.deposit(fromAcc, -amount)

            self.deposit(toAcc, amount)

            return True

import threading

from typing import List, Any



class BoundedBlockingBuffer:

    """Thread-safe bounded buffer using Condition variable."""

    def __init__(self, capacity: int) -> None:

        self.capacity = capacity

        self.queue: List[Any] = []

        self.lock = threading.Lock()

        self.notFull = threading.Condition(self.lock)

        self.notEmpty = threading.Condition(self.lock)



    def put(self, item: Any) -> None:

        with self.notFull:

            while len(self.queue) >= self.capacity:

                self.notFull.wait()  # Releases lock and sleeps until notified

            self.queue.append(item)

            self.notEmpty.notify()  # Signal waiting consumers



    def get(self) -> Any:

        with self.notEmpty:

            while len(self.queue) == 0:

                self.notEmpty.wait()  # Releases lock and sleeps until notified

            item = self.queue.pop(0)

            self.notFull.notify()   # Signal waiting producers

            return item

import time

import threading

from abc import ABC, abstractmethod



class IEvictionStrategy(ABC):

    @abstractmethod

    def recordAccess(self, key: str) -> None: ...

    @abstractmethod

    def recordWrite(self, key: str) -> None: ...

    @abstractmethod

    def evictKey(self) -> Optional[str]: ...



class CacheEntry:

    __slots__ = ("value", "expiresAt")

    def __init__(self, value: Any, ttlSeconds: Optional[float] = None) -> None:

        self.value = value

        self.expiresAt = (time.time() + ttlSeconds) if ttlSeconds else None



    def isExpired(self) -> bool:

        return self.expiresAt is not None and time.time() > self.expiresAt



class ThreadSafeCacheEngine:

    """Thread-safe In-Memory Cache with pluggable Eviction Policy and TTL."""

    def __init__(self, capacity: int, evictionPolicy: IEvictionStrategy) -> None:

        self.capacity = capacity

        self.evictionPolicy = evictionPolicy

        self._lock = threading.RLock()

        self._storage: Dict[str, CacheEntry] = {}



    def get(self, key: str) -> Optional[Any]:

        with self._lock:

            entry = self._storage.get(key)

            if entry is None:

                return None

            if entry.isExpired():

                del self._storage[key]

                return None

            self.evictionPolicy.recordAccess(key)

            return entry.value



    def put(self, key: str, value: Any, ttlSeconds: Optional[float] = None) -> None:

        with self._lock:

            if key not in self._storage and len(self._storage) >= self.capacity:

                evictedKey = self.evictionPolicy.evictKey()

                if evictedKey and evictedKey in self._storage:

                    del self._storage[evictedKey]



            self._storage[key] = CacheEntry(value, ttlSeconds)

            self.evictionPolicy.recordWrite(key)

🚀 Time-To-Live (TTL) Caching

Combine lazy on-read cleanup with daemon active purging to guarantee zero memory leakage and constant-time reads.

Dual TTL Expiration Mechanics (Lazy + Active Daemon)

Use when: handling key expiration cleanly without leaking memory on unread keys or stalling active reads.

  ├── Yes -> Delete & return None (Lazy O(1))
  └── No  -> Return value

Background Daemon Thread (Every 1s):
  └── Scans storage & evicts unread expired keys (Active)
import time

import threading

from typing import Dict, Any, Optional



class TtlCacheEngine:

    """In-memory cache with dual eviction: Lazy on-read + Active background daemon."""

    def __init__(self, cleanIntervalSeconds: float = 1.0) -> None:

        self._lock = threading.RLock()

        self._storage: Dict[str, tuple[Any, float]] = {}  # key -> (value, expiryEpoch)

        self._isRunning = True



        # Active background cleanup thread

        self._cleanerThread = threading.Thread(target=self._activeCleanupLoop, daemon=True)

        self._cleanerThread.start()



    def put(self, key: str, value: Any, ttlSeconds: float) -> None:

        with self._lock:

            self._storage[key] = (value, time.time() + ttlSeconds)



    def get(self, key: str) -> Optional[Any]:

        with self._lock:

            item = self._storage.get(key)

            if item is None:

                return None

            val, expiry = item

            if time.time() > expiry:

                # Lazy cleanup on read

                del self._storage[key]

                return None

            return val



    def _activeCleanupLoop(self) -> None:

        while self._isRunning:

            time.sleep(1.0)

            now = time.time()

            with self._lock:

                expiredKeys = [k for k, (_, exp) in self._storage.items() if now > exp]

                for k in expiredKeys:

                    del self._storage[k]

O(1) LRU Eviction via Doubly Linked List

Use when: implementing the canonical Least Recently Used eviction algorithm from scratch in machine coding rounds.

Architecture & Design Trace

Combining dummy sentinel head/tail nodes with a key-to-node dictionary enables O(1) removal, insertion at head, and eviction from tail.

Head <--> [Key 3] <--> [Key 1] <--> [Key 2] <--> Tail
  ▲                                             ▲
  └── Most Recently Used                Least Recently Used (Evicted!)
from typing import Optional, Dict



class Node:

    __slots__ = ("key", "prev", "next")

    def __init__(self, key: str = "") -> None:

        self.key = key

        self.prev: Optional["Node"] = None

        self.next: Optional["Node"] = None



class LruEvictionPolicy(IEvictionPolicy):

    """O(1) LRU eviction using custom Doubly-Linked List + Hash Map."""

    def __init__(self) -> None:

        self.nodeMap: Dict[str, Node] = {}

        # Sentinel dummy head and tail

        self.head, self.tail = Node(), Node()

        self.head.next, self.tail.prev = self.tail, self.head



    def _removeNode(self, node: Node) -> None:

        node.prev.next, node.next.prev = node.next, node.prev



    def _insertAtHead(self, node: Node) -> None:

        node.next, node.prev = self.head.next, self.head

        self.head.next.prev, self.head.next = node, node



    def recordAccess(self, key: str) -> None:

        if key in self.nodeMap:

            node = self.nodeMap[key]

            self._removeNode(node)

            self._insertAtHead(node)



    def recordWrite(self, key: str) -> None:

        if key in self.nodeMap:

            self.recordAccess(key)

        else:

            newNode = Node(key)

            self.nodeMap[key] = newNode

            self._insertAtHead(newNode)



    def evict(self) -> Optional[str]:

        if self.tail.prev == self.head:

            return None  # Empty list

        lruNode = self.tail.prev

        self._removeNode(lruNode)

        del self.nodeMap[lruNode.key]

        return lruNode.key

🏛️ Transactional Key-Value Store

Deliver a complete, thread-safe Key-Value store supporting nested transactions (BEGIN, COMMIT, ROLLBACK) and tombstones.

Transactional In-Memory Key-Value Store

Use when: solving one of the most frequently asked machine coding questions (In-Memory Key-Value Store with Transactions).

Architecture & Design Trace

Nested transactions are maintained using a stack of mutation frames. Reads inspect the stack top-down to find the most recent value or tombstone. Rollbacks simply pop the stack.

Global Store: {"a": 10, "b": 20}
Stack:
  Tx 1 (Parent): {"a": 99}
  Tx 2 (Child) : {"b": None} (Tombstone deletion)
-----------------------------------------------
get("b") -> None (Shadowed by Tx 2)
get("a") -> 99   (Shadowed by Tx 1)
rollback() -> Pops Tx 2 -> get("b") returns 20!
import threading

from typing import Dict, Any, Optional, List



class TransactionState:

    def __init__(self) -> None:

        self.mutations: Dict[str, Optional[Any]] = {}  # key -> value (None = deleted)



class TransactionalKeyValueStore:

    """Thread-safe Key-Value Store supporting nested transactions (BEGIN, COMMIT, ROLLBACK)."""

    def __init__(self) -> None:

        self._lock = threading.RLock()

        self._globalStore: Dict[str, Any] = {}

        self._transactionStack: List[TransactionState] = []



    def begin(self) -> int:

        """Starts a new transaction layer. Returns current nesting depth."""

        with self._lock:

            self._transactionStack.append(TransactionState())

            return len(self._transactionStack)



    def put(self, key: str, value: Any) -> None:

        with self._lock:

            if self._transactionStack:

                # Stage in current active transaction

                self._transactionStack[-1].mutations[key] = value

            else:

                self._globalStore[key] = value



    def get(self, key: str) -> Optional[Any]:

        with self._lock:

            # Check transaction stack from innermost to outermost

            for tx in reversed(self._transactionStack):

                if key in tx.mutations:

                    return tx.mutations[key]

            return self._globalStore.get(key)



    def delete(self, key: str) -> bool:

        with self._lock:

            currentVal = self.get(key)

            if currentVal is None:

                return False

            if self._transactionStack:

                self._transactionStack[-1].mutations[key] = None  # Tombstone

            else:

                del self._globalStore[key]

            return True



    def commit(self) -> bool:

        """Commits the active transaction layer."""

        with self._lock:

            if not self._transactionStack:

                return False

            activeTx = self._transactionStack.pop()

            if self._transactionStack:

                # Merge into parent transaction

                self._transactionStack[-1].mutations.update(activeTx.mutations)

            else:

                # Apply directly to global store

                for k, v in activeTx.mutations.items():

                    if v is None:

                        self._globalStore.pop(k, None)

                    else:

                        self._globalStore[k] = v

            return True



    def rollback(self) -> bool:

        """Aborts active transaction layer, discarding staged changes."""

        with self._lock:

            if not self._transactionStack:

                return False

            self._transactionStack.pop()

            return True

🏛️ Multi-Strategy Rate Limiter

Build a thread-safe rate limiter supporting Token Bucket and Sliding Window algorithms with the Strategy pattern.

Multi-Strategy Rate Limiter (Token Bucket & Sliding Window)

Use when: implementing API rate limiters with swappable algorithms (Token Bucket vs Sliding Window Log).

Architecture & Design Trace

Strategy pattern decouples the algorithm from the thread-safe coordinator. Token Bucket is memory-efficient O(1); Sliding Window Log guarantees exact boundary accuracy.

Incoming Request -> ThreadSafeRateLimiter (RLock)
                          │
                          └── delegates to <<IRateLimitStrategy>>
                                 ├── TokenBucketStrategy (O(1) memory)
                                 └── SlidingWindowLogStrategy (Exact boundaries)
import time

import threading

from abc import ABC, abstractmethod

from typing import Dict, List



class IRateLimitStrategy(ABC):

    @abstractmethod

    def allowRequest(self, key: str, maxRequests: int, windowSeconds: float) -> bool: ...



class TokenBucketStrategy(IRateLimitStrategy):

    """Token Bucket algorithm with continuous refill."""

    def __init__(self) -> None:

        self.buckets: Dict[str, tuple[float, float]] = {}  # key -> (tokens, lastRefillTime)



    def allowRequest(self, key: str, maxRequests: int, windowSeconds: float) -> bool:

        now = time.time()

        refillRate = maxRequests / windowSeconds

        if key not in self.buckets:

            self.buckets[key] = (float(maxRequests) - 1.0, now)

            return True



        tokens, lastTime = self.buckets[key]

        tokens = min(float(maxRequests), tokens + (now - lastTime) * refillRate)



        if tokens >= 1.0:

            self.buckets[key] = (tokens - 1.0, now)

            return True

        self.buckets[key] = (tokens, now)

        return False



class SlidingWindowLogStrategy(IRateLimitStrategy):

    """Sliding Window Log algorithm: Exact timestamp tracking."""

    def __init__(self) -> None:

        self.logs: Dict[str, List[float]] = {}



    def allowRequest(self, key: str, maxRequests: int, windowSeconds: float) -> bool:

        now = time.time()

        cutoff = now - windowSeconds

        timestamps = self.logs.setdefault(key, [])

        # Prune older entries

        while timestamps and timestamps[0] <= cutoff:

            timestamps.pop(0)



        if len(timestamps) < maxRequests:

            timestamps.append(now)

            return True

        return False



class ThreadSafeRateLimiter:

    """Orchestrator wrapping rate limit strategy with reentrant thread safety."""

    def __init__(self, strategy: IRateLimitStrategy) -> None:

        self.strategy = strategy

        self._lock = threading.RLock()



    def isAllowed(self, key: str, maxRequests: int, windowSeconds: float) -> bool:

        with self._lock:

            return self.strategy.allowRequest(key, maxRequests, windowSeconds)
